```kotlin
private fun doPush() {
    val backend = selectedBackend ?: return
    // (remote confirmation dialog — NOT needed for pull)
    ApplicationManager.getApplication().executeOnPooledThread {
        val result = backend.push()
        ApplicationManager.getApplication().invokeLater {
            if (disposed) return@invokeLater
            if (!result.isSuccess) {
                Notifications.Bus.notify(Notification(
                    "SystemExplorer", "Push Failed",
                    result.stderr.takeLast(300).ifBlank { "Unknown error" },
                    NotificationType.ERROR
                ), project)
            } else {
                reloadData()
            }
        }
    }
}
```

    Step 4: Add doPull() private method in the "Write operations" section, after doPush():

    ```kotlin
    private fun doPull() {
        val backend = selectedBackend ?: return
        if (pullInProgress) return
        pullInProgress = true

        ProgressManager.getInstance().run(object : Task.Backgroundable(project, "Pulling\u2026", false) {
            override fun run(indicator: com.intellij.openapi.progress.ProgressIndicator) {
                val result = backend.pull()
                ApplicationManager.getApplication().invokeLater {
                    pullInProgress = false   // ALWAYS reset before disposed check
                    if (disposed) return@invokeLater
                    when {
                        !result.isSuccess -> {
                            Notifications.Bus.notify(
                                Notification(
                                    "SystemExplorer",
                                    "Pull Failed",
                                    result.stderr.takeLast(300).ifBlank { "Unknown error" },
                                    NotificationType.ERROR
                                ), project
                            )
                        }
                        result.stdout.contains("CONFLICT") -> {
                            Notifications.Bus.notify(
                                Notification(
                                    "SystemExplorer",
                                    "Pull Conflicts",
                                    "Pull resulted in conflicts \u2014 resolve the marked files",
                                    NotificationType.WARNING
                                ), project
                            )
                            reloadData()
                        }
                        else -> reloadData()
                    }
                }
            }
        })
    }
    ```

    IMPORTANT: The `canBeCancelled = false` parameter (third argument to Task.Backgroundable constructor) prevents a fake Cancel button appearing in the progress indicator. The existing git executor has no cooperative cancellation — do not add a cancel UI that does nothing.

    IMPORTANT: `pullInProgress = false` must be the FIRST statement inside `invokeLater`, before `if (disposed) return@invokeLater`. This prevents the Pull button from staying permanently disabled if the panel is disposed mid-pull and then recreated.
