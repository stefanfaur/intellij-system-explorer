```kotlin
private val HARD_EXCLUDE_DIRS = setOf("node_modules", ".git", "build", "target", ".gradle", ".idea", "__pycache__", ".DS_Store", "dist", "out")
```

**`countFiles(root: Path, timeoutMs: Long = 5_000): Int`** — runs on Dispatchers.IO, uses `Files.walk(root).use { it.filter(Files::isRegularFile).count().toInt() }` with a 5-second timeout. Returns -1 on timeout or error.

**`shouldIndex(file: Path, allowedExts: Set<String>, maxSizeBytes: Long = 1_048_576): Boolean`**:
1. Skip if extension not in allowedExts (extract extension from file name, lowercase, no dot)
2. Skip if file size > maxSizeBytes
3. Skip if file starts with binary content: read first 8 bytes and check for bytes outside ASCII printable range (0x00–0x08 range = binary); a simple heuristic is: if any of first 8 bytes is 0x00, it's binary

**`buildIndex(root: Path, manager: LuceneIndexManager, settings: QuickOpenSettings.State)`** — suspend function, must be called from Dispatchers.IO:
1. Parse `settings.luceneExtensionAllowlist` into a `Set<String>` by splitting on ","
2. Calculate `maxSizeBytes = settings.luceneMaxIndexSizeMb.toLong() * 1024 * 1024`
3. Track `totalIndexedBytes: Long` — when it exceeds the per-root max, flip a flag `contentCapReached = true`; continue indexing paths but pass `allowContent = false`
4. Use `Files.walkFileTree(root, ...)` with a `SimpleFileVisitor<Path>`:
   - `preVisitDirectory`: return `SKIP_SUBTREE` if the directory name is in HARD_EXCLUDE_DIRS, or if `!settings.ripgrepSearchHidden` and name starts with "."
   - `visitFile`: call `shouldIndex(file, allowedExts, maxSizeBytes)`. If true, read content safely (`file.toFile().readText(Charsets.UTF_8)` wrapped in try/catch returning null on any exception), call `manager.addOrUpdateFile(file, content, !contentCapReached)`, update totalIndexedBytes
5. Call `manager.commit()` after the walk completes
6. Log the count with `LOG.info("Indexed ${manager.numDocs()} docs for $root")`

For the test, use a temp directory created with `Files.createTempDirectory`. Create real files. Verify via `manager.searchPaths(...)` returning expected results.

Add `startWatcher(root: Path, manager: LuceneIndexManager, settings: QuickOpenSettings.State, scope: CoroutineScope)` to LuceneIndexBuilder.

This is a suspend function that must run on Dispatchers.IO. It should NOT block a thread permanently — use `withContext(Dispatchers.IO)` for the blocking `watchService.poll()` call.

**Implementation:**
```kotlin
suspend fun startWatcher(root: Path, manager: LuceneIndexManager, settings: QuickOpenSettings.State, scope: CoroutineScope) {
    val allowedExts = settings.luceneExtensionAllowlist.split(",").map { it.trim().lowercase() }.toSet()
    FileSystems.getDefault().newWatchService().use { watchService ->
        // Register all existing subdirectories
        Files.walkFileTree(root, object : SimpleFileVisitor<Path>() {
            override fun preVisitDirectory(dir: Path, attrs: BasicFileAttributes): FileVisitResult {
                if (dir.fileName?.toString() in HARD_EXCLUDE_DIRS) return FileVisitResult.SKIP_SUBTREE
                if (!settings.ripgrepSearchHidden && dir.fileName?.toString()?.startsWith(".") == true && dir != root) return FileVisitResult.SKIP_SUBTREE
                dir.register(watchService, ENTRY_CREATE, ENTRY_MODIFY, ENTRY_DELETE)
                return FileVisitResult.CONTINUE
            }
        })
        while (scope.isActive) {
            val key = withContext(Dispatchers.IO) {
                watchService.poll(500, TimeUnit.MILLISECONDS)
            } ?: continue
            for (event in key.pollEvents()) {
                when (event.kind()) {
                    StandardWatchEventKinds.OVERFLOW -> {
                        // Events were lost — full re-scan to close the staleness gap
                        scope.launch(Dispatchers.IO) { buildIndex(root, manager, settings) }
                        key.reset()
                        break
                    }
                    StandardWatchEventKinds.ENTRY_DELETE -> {
                        val changed = (key.watchable() as Path).resolve(event.context() as Path)
                        manager.deleteFile(changed.toString())
                    }
                    StandardWatchEventKinds.ENTRY_CREATE,
                    StandardWatchEventKinds.ENTRY_MODIFY -> {
                        val changed = (key.watchable() as Path).resolve(event.context() as Path)
                        if (Files.isDirectory(changed)) {
                            // Register new subdirectory so its contents are watched too
                            if (changed.fileName?.toString() !in HARD_EXCLUDE_DIRS) {
                                changed.register(watchService, ENTRY_CREATE, ENTRY_MODIFY, ENTRY_DELETE)
                            }
                        } else if (shouldIndex(changed, allowedExts)) {
                            val content = try { changed.toFile().readText(Charsets.UTF_8) } catch (_: Exception) { null }
                            manager.addOrUpdateFile(changed, content, content != null)
                        }
                    }
                }
            }
            manager.commit()
            if (!key.reset()) break
        }
    }
}
```

Import `java.nio.file.StandardWatchEventKinds.*` at the top of the file for cleaner code.

Do NOT register the WatchService on the full tree eagerly — the `Files.walkFileTree` registration above is acceptable because it happens once at watcher startup (after buildIndex has already walked the tree). The anti-pattern to avoid is registering within the critical hot path of event processing.
