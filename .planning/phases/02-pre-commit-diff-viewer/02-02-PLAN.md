  <action>
    Create a new file `InlineDiffPanel.kt` in the `ro.faur.explorer.gitpanel.ui` package.

    InlineDiffPanel is a JPanel that owns the child disposable lifecycle for embedded DiffRequestPanel instances. It shows a spinner while loading, replaces it with the diff viewer when ready, and clears on demand.

    Key design points:
    - Constructor takes `project: Project` and `parentDisposable: Disposable`
    - Internal state: `var currentChildDisposable: Disposable? = null`
    - `showSpinner()`: removes all components, adds a centered `AsyncProcessIcon("diff-loading")` wrapped in a BorderLayout JPanel
    - `clear()`: disposes currentChildDisposable (null it out), calls removeAll() + revalidate()
    - `showDiffRequest(request: SimpleDiffRequest)`:
        1. Disposes currentChildDisposable if non-null
        2. Creates new child: `Disposer.newDisposable(parentDisposable)`
        3. Stores as currentChildDisposable
        4. `DiffManager.getInstance().createRequestPanel(project, childDisposable, null)`
        5. `panel.setRequest(request)`
        6. removeAll(), add(panel.component, BorderLayout.CENTER), revalidate(), repaint()

    CRITICAL: `showSpinner()` and `showDiffRequest()` must be called on EDT. Enforce via assertion or doc comment. Do NOT call DiffManager methods off-EDT.

    InlineDiffPanel itself should be a `JPanel(BorderLayout())` — it acts as a container slot in the GitPanelComponent layout.

    Imports needed: `com.intellij.diff.*`, `com.intellij.openapi.util.Disposer`, `com.intellij.util.ui.AsyncProcessIcon`, `java.awt.BorderLayout`, `javax.swing.JPanel`.
  </action>

    private fun loadInlineDiff(commitFile: CommitFile) {
        val backend = selectedBackend ?: return
        val snapshotKey = System.identityHashCode(backend)

        inlineDiffPanel.showSpinner()

        ApplicationManager.getApplication().executeOnPooledThread {
            val headBytes: ByteArray? = when (commitFile.status) {
                GitFileStatus.UNTRACKED, GitFileStatus.ADDED -> null
                else -> backend.getHeadContent(commitFile.path)
            }
            val workBytes: ByteArray? = when (commitFile.status) {
                GitFileStatus.DELETED -> null
                else -> runCatching {
                    java.io.File(backend.repoPath, commitFile.path).readBytes()
                }.getOrNull()
            }

            val request = buildDiffRequest(project, backend.repoPath, commitFile, headBytes, workBytes)

            ApplicationManager.getApplication().invokeLater {
                if (disposed) return@invokeLater
                if (System.identityHashCode(selectedBackend) != snapshotKey) return@invokeLater
                if (selectedDiffFile?.path != commitFile.path) return@invokeLater  // file switched

    **7. Implement buildDiffRequest helper (private):**
    ```kotlin
    private fun buildDiffRequest(
        project: Project,
        repoPath: String,
        commitFile: CommitFile,
        headBytes: ByteArray?,
        workBytes: ByteArray?
    ): SimpleDiffRequest {
        val factory = DiffContentFactory.getInstance()
        val filePath = VcsUtil.getFilePath(java.io.File(repoPath, commitFile.path))

        val headContent: DiffContent = if (headBytes != null)
            factory.createDocumentFromBytes(project, headBytes, filePath)
        else
            factory.createEmpty()

        val workContent: DiffContent = if (workBytes != null)
            factory.createDocumentFromBytes(project, workBytes, filePath)
        else
            factory.createEmpty()

        return SimpleDiffRequest(
            commitFile.path,
            headContent,
            workContent,
            "HEAD",
            "Working Tree"
        )
    }
    ```
