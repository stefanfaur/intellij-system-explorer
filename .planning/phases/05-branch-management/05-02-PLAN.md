// JBPopupFactory for branch list popup:
import com.intellij.openapi.ui.popup.JBPopupFactory
import com.intellij.ui.awt.RelativePoint

val popup = JBPopupFactory.getInstance()
    .createPopupChooserBuilder(branches)    // branches: List<BranchInfo>
    .setTitle("Switch Branch")
    .setItemChosenCallback { branch: BranchInfo ->
        if (!branch.isCurrent) doCheckoutBranch(branch.name)
    }
    .setRenderer(object : com.intellij.ui.ColoredListCellRenderer<BranchInfo>() {
        override fun customizeCellRenderer(list: JList<out BranchInfo>, value: BranchInfo, index: Int, selected: Boolean, hasFocus: Boolean) {
            if (value.isCurrent) append("* ", com.intellij.ui.SimpleTextAttributes.REGULAR_BOLD_ATTRIBUTES)
            append(value.name, com.intellij.ui.SimpleTextAttributes.REGULAR_ATTRIBUTES)
        }
    })
    .createPopup()
popup.showUnderneathOf(branchLabel)

// Three-option dialog (Messages.showDialog):

)

  <behavior>
    - doCheckoutBranch("feature") with clean working tree → calls backend.checkoutBranch("feature") on pooled thread → invokeLater calls reloadData() on success
    - doCheckoutBranch("feature") with dirty working tree → shows Messages.showDialog with 3 options; user clicks Cancel → no checkout, no stash, no reset
    - doCheckoutBranch("feature") with dirty working tree → user clicks "Stash & Switch" → stash(msg="Auto-stash before checkout to feature", includeUntracked=true) called on pooled thread → then checkoutBranch("feature") → then reloadData()
    - doCheckoutBranch("feature") with dirty working tree → user clicks "Discard & Switch" → resetHard() called on pooled thread → then checkoutBranch("feature") → then reloadData()
    - On checkout failure → Notifications.Bus.notify with NotificationType.ERROR
  </behavior>
  <action>
    Add four private methods to GitPanelComponent:

    **doCheckoutBranch(targetBranch: String):**
    - Guard: `val backend = selectedBackend ?: return`
    - `val snapshotKey = System.identityHashCode(backend)`
    - `executeOnPooledThread { val status = backend.getWorkingTreeStatus(); invokeLater { if (disposed) return; if stale return; if status.isNotEmpty() showDirtyTreeDialog(backend, targetBranch, snapshotKey) else performCheckout(backend, targetBranch, snapshotKey) } }`

    **showDirtyTreeDialog(backend, targetBranch, snapshotKey):** (called on EDT)
    - Must run on EDT (already guaranteed by caller's invokeLater)
    - `val result = Messages.showDialog(project, "You have uncommitted changes in '$targetBranch'. What would you like to do?", "Dirty Working Tree", arrayOf("Stash & Switch", "Discard & Switch", "Cancel"), 2, Messages.getWarningIcon())`
    - `when (result) { 0 -> doStashAndCheckout(backend, targetBranch, snapshotKey); 1 -> doDiscardAndCheckout(backend, targetBranch, snapshotKey); else -> return }`

    **doStashAndCheckout(backend, targetBranch, snapshotKey):** (called on EDT)
    - `executeOnPooledThread { val stashResult = backend.stash("Auto-stash before checkout to $targetBranch", includeUntracked=true); if (!stashResult.isSuccess) { invokeLater { notifyError("Stash failed: ${stashResult.stderr.takeLast(200)}") }; return@executeOnPooledThread }; performCheckout(backend, targetBranch, snapshotKey) }`

    **doDiscardAndCheckout(backend, targetBranch, snapshotKey):** (called on EDT)
    - `executeOnPooledThread { val resetResult = backend.resetHard(); if (!resetResult.isSuccess) { invokeLater { notifyError("Reset failed: ${resetResult.stderr.takeLast(200)}") }; return@executeOnPooledThread }; performCheckout(backend, targetBranch, snapshotKey) }`

    **performCheckout(backend, targetBranch, snapshotKey):** (called on pooled thread)
    - `val result = backend.checkoutBranch(targetBranch); invokeLater { if (disposed || stale) return; if (result.isSuccess) reloadData() else notifyError("Checkout failed: ${result.stderr.takeLast(300)}") }`

    Helper: add `private fun notifyError(msg: String)` that calls `Notifications.Bus.notify(Notification("SystemExplorer", "Branch Error", msg, NotificationType.ERROR), project)` — matching the notification group already used in the file.

    Note: The dialog message should make "Discard & Switch" clearly destructive — use wording like "Discard & Switch (WARNING: all uncommitted changes will be permanently lost)".
