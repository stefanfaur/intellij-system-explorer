```kotlin
import javax.swing.JComponent

class CreateBranchDialog(project: Project) : DialogWrapper(project, true) {
    private val nameField = JBTextField(30)
    private val errorLabel = JBLabel("").apply {
        foreground = com.intellij.ui.JBColor.RED
    }

    init {
        title = "Create New Branch"
        init()  // MUST call init() in init block after setting title
        nameField.document.addDocumentListener(object : com.intellij.openapi.editor.event.DocumentListener {
            // Use javax.swing DocumentListener via DocumentAdapter or direct implementation:
        })
    }

    override fun createCenterPanel(): JComponent {
        return FormBuilder.createFormBuilder()
            .addLabeledComponent("Branch name:", nameField)
            .addComponent(errorLabel)
            .panel
    }

    // isOKActionEnabled = false/true controls the OK button
    fun getBranchName(): String = nameField.text.trim()
}
```

```kotlin
private fun showDeleteBranchSelectPopup() {
    val deletable = cachedBranches.filter { !it.isCurrent }
    if (deletable.isEmpty()) {
        Messages.showInfoMessage(project, "No other branches to delete.", "Delete Branch")
        return
    }
    val popup = JBPopupFactory.getInstance()
        .createPopupChooserBuilder(deletable)
        .setTitle("Select Branch to Delete")
        .setItemChosenCallback { branch -> doDeleteBranch(branch.name) }
        .createPopup()
    // anchor: find the delete button's component, or use a stored reference
    popup.showUnderneathOf(deleteButton)  // deleteButton is a stored JButton/ActionButton ref
}
```
